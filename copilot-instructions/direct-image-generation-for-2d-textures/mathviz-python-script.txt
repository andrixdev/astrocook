# Generate 2D or 3D textures for Unity out of math stuff

import math
import random
import numpy as np
import OpenEXR
import datetime

# Create cloud
def create_cloud (size):
    cloud =  np.zeros((size, 3)).astype('float32')
    
    print("Creating a cloud of size " + str(size))
    
    for i in range(0, size):
        x = random.random()
        y = random.random()
        z = random.random()
        cloud[i] = (x, y, z)
    
    return cloud

# p is dimension of distance
def get_sum_of_distances(point, cloud, p):
    size = cloud.shape[0]
    
    sum_of_dists = 0
    
    for i in range(0, size):
        cloud_point = cloud[i]
        dx = abs(point[0] - cloud_point[0])
        dy = abs(point[1] - cloud_point[1])
        dz = abs(point[2] - cloud_point[2])
        sum_of_dists += ((dx ** p) + (dy ** p) + (dz ** p)) ** (1/p)
    
    # print("Sum of dists is " + str(sum_of_dists))
    
    return sum_of_dists
    
def get_cloud_center (cloud, distance_power, candidates_dimension, candidates_confine):
    start_time = datetime.datetime.now()
    
    dim = cloud.shape[0]
    candim = candidates_dimension
    pop = candim ** 3
    p = distance_power
    
    # Spawn grid of candidates
    # for each candidate scan whole cloud to get sum of distances
    wannabes = np.zeros((candim, candim, candim))
    
    min_dist = 99999999
    wannabe = [0, 0, 0]
    start = math.floor(candim / 4) if candidates_confine else 0
    end = math.floor(3 * candim / 4) if candidates_confine else candim
    
    for i in range(start, end):
        for j in range(start, end):
            for k in range(start, end):
                x = i / candim
                y = j / candim
                z = k / candim
                dist = get_sum_of_distances((x, y, z), cloud, p)
                if (dist < min_dist):
                    min_dist = dist
                    wannabe = [x, y, z]
                
    end_time = datetime.datetime.now()
    time = end_time.timestamp() - start_time.timestamp()
    # print("Sums of distances for p = " + str(round(p, 3)) + " is minimal for " + str(min_dist) + " won by " + str(wannabe) + " in " + str(round(time, 2)) + "s")
    print("Sums of distances for p = " + str(round(p, 3)) + " won by " + str(wannabe) + " in " + str(round(time, 2)) + "s")
    
    return wannabe
    
def imagify_cloud (cloud):
    # Compute required img size
    dim = cloud.shape[0]
    pixel_nb = dim
    img_size = math.ceil(pixel_nb ** (1/2))
    img =  np.zeros((img_size, img_size, 3)).astype('float32')
    
    count = 0
    for i in range(0, pixel_nb):
        img_x_pos = count % img_size
        img_y_pos = math.floor(count / img_size)
        cloud_point = cloud[i]
        img[img_x_pos][img_y_pos] = (cloud_point[0], cloud_point[1], cloud_point[2])
        count += 1
        
    return img
    
def imagify_centers (centers):
    pixel_nb = centers.shape[0]
    img_size = math.ceil(pixel_nb ** (1/2))
    img =  np.zeros((img_size, img_size, 3)).astype('float32')
    
    count = 0
    for i in range(0, pixel_nb):
        img_x_pos = count % img_size
        img_y_pos = math.floor(count / img_size)
        center = centers[i]
        img[img_x_pos][img_y_pos] = center
        count += 1
        
    return img
    
# Create cloud
population = 5
cloud = create_cloud(population)

# Generate candidates and compute centers
candidates_dimension = 200
candidates_confine = True
actual_candidates = math.floor((candidates_dimension ** 3) * (1/27 if candidates_confine else 1))
print("Proposing " + str(actual_candidates) + " center candidates...")
norms = 50
print("Exploring " + str(norms) + " different norm definitions...")
centers = np.zeros((norms, 3))

for n in range(1, norms):
    norm = 1 + 0.01 * (n ** 2)
    center = get_cloud_center(cloud, norm, candidates_dimension, candidates_confine)
    centers[n-1] = center
    
cloud_img = imagify_cloud(cloud)
centers_img = imagify_centers(centers)

cloud_img_size = population
centers_img_size = math.ceil(norms ** (1/2))
confined_str = "-confined" if candidates_confine else ""

# Generate cloud EXR file
channels = { "RGB" : cloud_img }
header = { "compression" : OpenEXR.NO_COMPRESSION,
           "type" : OpenEXR.scanlineimage }
with OpenEXR.File(header, channels) as outfile:
    outfile.write("./output/cloud-" + str(cloud_img_size) + "-" + str(math.floor(datetime.datetime.now().timestamp())) + ".exr")

# Generate centers EXR file
channels = { "RGB" : centers_img }
with OpenEXR.File(header, channels) as outfile:
    outfile.write("./output/centers-" + str(norms) + "-" + str(math.floor(datetime.datetime.now().timestamp())) + ".exr")
