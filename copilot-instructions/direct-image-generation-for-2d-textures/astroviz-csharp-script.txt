 * ANDRIX ® 2025-2026 🤙
 * 
 * This script generates 2D textures out of text data containing rows
 *
 * All values have to be separated by a space character
 * End of lines are needed too ("\n")
 *
 * The outputted textures can be sampled in Unity VFX graph to retrieve information
 * Once they are in the Assets, don't forget to import them with the right parameters (see README-texture-generator.txt)
 * 
 */

		static string PreprendZeros(int input, int targetLength)
		{
			string result = input.ToString();
			int size = result.Length;
			for (int i = 0; i < targetLength - size; i++)
			{
				result = "0" + result;
			}
			return result;
		}

		[MenuItem("Astroyum/Scan for max texture size of Dwarfgal")]
		static void ScanForMaxTextureSizeDwarfgal()
		{
			int maxSize = 0;

			for (int f = 1; f <= 500; f++)
			{
				// Build file name
				string frameName = PreprendZeros(f, 3);
				string sourceTextFileName = "dwarfgal/500-frames-1-in-3/dwarfgal-xyz-" + frameName + "-1-in-3";
				string sourceTextPath = Application.dataPath + "/Textures2D/ParticlesTextData/" + sourceTextFileName + ".txt";

				int size = GetTextureSize(sourceTextPath);
				if (size > maxSize) {
					maxSize = size;
				}
			}

			Debug.Log("<color=#eeee22ff>[CreateTexture2DFromEditor] Max Dwarfgal texture size is " + maxSize + "</color>");

		}

		static int GetTextureSize(string sourceTextPath)
		{
			// Read file
			StreamReader reader = new StreamReader(sourceTextPath);
			string text = reader.ReadToEnd();
			reader.Close();
			string[] lines = text.Split('\n');

			// Compute texture size
			return (int) Mathf.Ceil(Mathf.Sqrt(lines.Length));//Mathf.Floor(Mathf.Sqrt(lines.Length));
		}

		// forcedSize is optional (defaults to 0, considered optional)
		// forcedSize artificially increases texture size with dummy empty colors to match others in a sequence
		static void CreateTexture2D(string sourceAuthorFolderToken, string sourceFramesFolderToken, string sourceFileName, float[] arrayOfMins, float[] arrayOfMaxs, int forcedSize = 0)
		{
			float time = Time.realtimeSinceStartup;
			
			// Check how many variables (volumns, values) we have other than xyz
			int remaining = Mathf.Max(0, arrayOfMins.Length - 3);
			int numberOfTextures = 1 + (int) Mathf.Ceil(remaining / 3.0f);
			Debug.Log("<color=#ff33ccff>[CreateTexture2DFromEditor] Data has " + arrayOfMins.Length + " columns. " + numberOfTextures + " textures with name '" + sourceFileName + "' will be generated.</color>");
			
			// Read the text directly from the source file
			string sourcePath = Application.dataPath + "/Textures2D/ParticlesTextData/" + sourceAuthorFolderToken + "/" + sourceFramesFolderToken + "/" + sourceFileName + ".txt";
			StreamReader reader = new StreamReader(sourcePath);
			string text = reader.ReadToEnd();
			reader.Close();
			
			string[] lines = text.Split('\n');
			
			// Compute size needed to store all data (loses a few points with Floor but makes sure all pixels are filled in a square, even if forcedSize is given)
			int len = lines.Length;
			int naturalSizeFloor = (int) Mathf.Floor(Mathf.Sqrt(len));
			int naturalSizeCeil = (int) Mathf.Ceil(Mathf.Sqrt(len));
			int size = naturalSizeCeil;
			if (forcedSize != 0) {
				if (forcedSize < size) {
					Debug.LogError("<color=#ff3333ff>[CreateTexture2DFromEditor] forcedSize is lower than naturalSize. Sir, this is bad.</color>");
				}
				size = forcedSize;
			}
			bool plural = numberOfTextures >= 2;
			Debug.Log("<color=#ff33ccff>[CreateTexture2DFromEditor] Generating " + numberOfTextures + " texture" + (plural ? "s" : "") + " of size " + size + "x" + size + " (== " + (size * size) + ") from " + lines.Length + " lines of text data...</color>");
			
			// Create arrays to store color data (3 variables per color array)
			// Array of color arrays
			Color[][] colorArrays = new Color[numberOfTextures][];
			Texture2D[] textures = new Texture2D[numberOfTextures];
			for (int co = 0; co < numberOfTextures; co++)
			{
				colorArrays[co] = new Color[size * size];
			}
			
			// Choose texture parameters
			TextureFormat format = TextureFormat.RGBAFloat;
			TextureWrapMode wrapMode = TextureWrapMode.Clamp;
			
			// Create the textures and apply the parameters
			for (int c = 0; c < numberOfTextures; c++)
			{
				Texture2D texture = new Texture2D(size, size, format, false);
				texture.wrapMode = wrapMode;
				textures[c] = texture;
			}
			
			// Get the 3 first elements (x, y and z)
			float xMin = arrayOfMins[0];
			float yMin = arrayOfMins[1];
			float zMin = arrayOfMins[2];
			float xMax = arrayOfMaxs[0];
			float yMax = arrayOfMaxs[1];
			float zMax = arrayOfMaxs[2];
			
			// Store other elements
			float[] mins = new float[remaining];
			float[] maxs = new float[remaining];
			for (int r = 0; r < remaining; r++)
			{
				mins[r] = arrayOfMins[3 + r];
				maxs[r] = arrayOfMaxs[3 + r];
			}
			
			// Scan all lines (leave some colors unwritten after naturalSize² in case of forcedSize)
			for (int i = 0; i < len; i++)
			{
				string[] fields = lines[i].Split(" ");
				
				// Read
				float x = Convert.ToSingle(fields[0], CultureInfo.InvariantCulture);
				float y = Convert.ToSingle(fields[1], CultureInfo.InvariantCulture);
				float z = Convert.ToSingle(fields[2], CultureInfo.InvariantCulture);
				
				// Remap
				float r = math.remap(xMin, xMax, 0.0f, 1.0f, x);
				float g = math.remap(yMin, yMax, 0.0f, 1.0f, y);
				float b = math.remap(zMin, zMax, 0.0f, 1.0f, z);
				
				// Values other than xyz
				float[] otherValues = new float[remaining];
				for (int l = 0; l < remaining; l++)
				{
					// Read
					float val = Convert.ToSingle(fields[3 + l], CultureInfo.InvariantCulture);
					
					// Remap
					otherValues[l] = math.remap(mins[l], maxs[l], 0.0f, 1.0f, val);
				}
				
				// Fill color arrays
				// First one with x, y, z
				colorArrays[0][i] = new Color(r, g, b, 1.0f);
				// Others
				for (int l = 0; l < numberOfTextures - 1; l++)
				{
					// Fill with zeros on the last texture if no data (there should always have data on rr)
					float rr = otherValues[3 * l];
					float gg = (3 * l + 1) < remaining ? otherValues[3 * l + 1] : 0.0f; 
					float bb = (3 * l + 2) < remaining ? otherValues[3 * l + 2] : 0.0f; 
					colorArrays[1 + l][i] = new Color(rr, gg, bb, 1.0f);
				}
			}

			// Copy the color values to the textures
			for (int t = 0; t < numberOfTextures; t++)
			{
				textures[t].SetPixels(colorArrays[t]);
				
				// Apply the changes to the texture and upload the updated texture to the GPU
				textures[t].Apply();
			}
			
			// Save the textures to our Unity Project
			for (int miam = 0; miam < numberOfTextures; miam++)
			{
				byte[] bytes = ImageConversion.EncodeToEXR(textures[miam]);

				// If there are 2 textures or more, create new folders to separate them
				if (numberOfTextures >= 2) {
					string potentialFolderName = "dimensions-" + PreprendZeros(1 + 3 * miam, 2) + "-" + PreprendZeros(1 + 3 * miam + 1, 2) + "-" + PreprendZeros(1 + 3 * miam + 2, 2);

					// Check if folder exists, if not create it
					if (!System.IO.Directory.Exists(Application.dataPath + "/Textures2D/" + sourceAuthorFolderToken + "/" + sourceFramesFolderToken + "/" + potentialFolderName)) {
						System.IO.Directory.CreateDirectory(Application.dataPath + "/Textures2D/" + sourceAuthorFolderToken + "/" + sourceFramesFolderToken + "/" + potentialFolderName);
					}
					System.IO.File.WriteAllBytes(Application.dataPath + "/Textures2D/" + sourceAuthorFolderToken + "/" + sourceFramesFolderToken + "/" + potentialFolderName + "/" + sourceFileName + "-tex-" + size + "-" + (1 + miam) + ".exr", bytes);
				}

				// Otherwise just write in the main folder
				else {
					System.IO.File.WriteAllBytes(Application.dataPath + "/Textures2D/" + sourceAuthorFolderToken + "/" + sourceFramesFolderToken + "/" + sourceFileName + "-tex-" + size + ".exr", bytes);
				}
				
			}
			
			// Goodbye thanks
			Debug.Log("<color=#ff33ccff>[CreateTexture2DFromEditor] Created and saved texture" + (plural ? "s" : "")  + " in " + (Time.realtimeSinceStartup - time) + " seconds 👌</color>");
			
		}
		
	}
